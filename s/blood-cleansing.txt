  function mkPath(pts) {

  function pAt(P, s) {
    s = clamp(s, 0, P.L); let i = 1; while (i < P.c.length - 1 && P.c[i] < s) i++;
    const a = P.p[i - 1], b = P.p[i], d = P.c[i] - P.c[i - 1] || 1, k = (s - P.c[i - 1]) / d;
    PT.x = a[0] + (b[0] - a[0]) * k; PT.y = a[1] + (b[1] - a[1]) * k; PT.a = Math.atan2(b[1] - a[1], b[0] - a[0]); return PT;
  }
  function tracePath(P, r = 14) {

  }

      P.push({ k: g.k, x: rr(x0, x1), y: rr(box[1] + g.r + 4, box[3] - g.r - 4), vx: Math.cos(a) * sp, vy: Math.sin(a) * sp, r: g.r, small: g.small, rot: rand() * TAU, i: P.length });
    }
    return { box, memX, pores, P, crossed: 0 };
  }
  function stepSim(sim, dt, opts = {}) {
    const [x0, y0, x1, y1] = sim.box, mx = sim.memX, half = 6;
    for (const p of sim.P) {
      const sp = p.small ? 46 : 16;
      if (p.thru) {   // тесік арқылы өтіп жатыр
        p.x += p.thru * sp * 1.3 * dt; p.y = lerp(p.y, p.py, Math.min(1, dt * 12));
        if ((p.thru > 0 && p.x > mx + half + p.r + 1) || (p.thru < 0 && p.x < mx - half - p.r - 1)) { sim.crossed += p.thru; p.vx = p.thru * sp; p.thru = 0; }
        continue;
      }
      p.vx += (rand() - 0.5) * sp * 12 * dt; p.vy += (rand() - 0.5) * sp * 12 * dt;
      if (opts.bias && p.small && p.k !== 'w') p.vx += opts.bias * dt;
      const v = hyp(p.vx, p.vy) || 1, k = sp / v; p.vx = lerp(p.vx, p.vx * k, 0.08); p.vy = lerp(p.vy, p.vy * k, 0.08);
      let nx = p.x + p.vx * dt, ny = p.y + p.vy * dt;
      if (ny < y0 + p.r) { ny = y0 + p.r; p.vy = Math.abs(p.vy); } if (ny > y1 - p.r) { ny = y1 - p.r; p.vy = -Math.abs(p.vy); }
      if (nx < x0 + p.r) { nx = x0 + p.r; p.vx = Math.abs(p.vx); } if (nx > x1 - p.r) { nx = x1 - p.r; p.vx = -Math.abs(p.vx); }
      const left = p.x < mx, lim = left ? mx - half - p.r : mx + half + p.r;
      if ((left && nx > lim) || (!left && nx < lim)) {
        let pore = null; if (p.small) for (const py of sim.pores) if (Math.abs(ny - py) < 11) { pore = py; break; }
        if (pore !== null && !(opts.oneWay && !left)) { p.thru = left ? 1 : -1; p.py = pore; nx = lim; }
        else { nx = lim; p.vx = -p.vx; }
      }
      p.x = nx; p.y = ny; p.rot += dt * 0.6;
    }
  }

  }

  function solveFlow(counter, N = 48) {
    const Cb = new Float64Array(N), Cd = new Float64Array(N), k = 0.085;
    for (let it = 0; it < 400; it++) {
      let c = 1; for (let i = 0; i < N; i++) { const tr = k * (c - Cd[i]); Cb[i] = c - tr; c = Cb[i]; }
      if (counter) { let d = 0; for (let i = N - 1; i >= 0; i--) { const tr = k * (Cb[i] - d) * 0.9; Cd[i] = d + tr; d = Cd[i]; } }
      else { let d = 0; for (let i = 0; i < N; i++) { const tr = k * (Cb[i] - d) * 0.9; Cd[i] = d + tr; d = Cd[i]; } }
    }
    return { Cb, Cd, N, out: Cb[N - 1] };
  }

  function hitAt(e) { const r = canvas.getBoundingClientRect(), x = e.clientX - r.left, y = e.clientY - r.top; for (let i = S.hits.length - 1; i >= 0; i--) { const h = S.hits[i]; if (x >= h.x && x <= h.x + h.w && y >= h.y && y <= h.y + h.h) return h; } return null; }
  canvas.addEventListener('pointermove', e => { const h = hitAt(e); S.hover = h ? h.s : null; canvas.style.cursor = h ? 'pointer' : 'default'; });
  canvas.addEventListener('pointerleave', () => { S.hover = null; });
  canvas.addEventListener('click', e => { const h = hitAt(e); if (h) toast(`<b>${h.s}</b> — ${h.info}`); });

  /* ---------- кадр ---------- */
  let last = performance.now();
  function frame(now) {
    const raw = Math.min(0.05, (now - last) / 1000); last = now;
    const dt = S.paused || REDUCED && false ? 0 : raw * S.speed;
    S.T += dt; S.fadeT = Math.min(1, S.fadeT + raw / 0.35);
    ctx = MAIN; const d = S.dpr;
    ctx.setTransform(d, 0, 0, d, 0, 0);
    ctx.fillStyle = lin(0, 0, 0, S.H, [[0, C.bg0], [1, C.bg1]]); ctx.fillRect(0, 0, S.W, S.H);
    ctx.setTransform(d * S.s, 0, 0, d * S.s, d * S.ox, d * S.oy);
    ctx.lineJoin = 'round';
    const st = ST[S.step];
    try { st.draw(S.T, dt); } catch (err) { console.error(err); }
    ctx.setTransform(d, 0, 0, d, 0, 0);
    drawLabels(st.lab ? st.lab(S.T) : null);
    if (S.fadeT < 1 && buf.width) { ctx.setTransform(1, 0, 0, 1, 0, 0); ctx.globalAlpha = 1 - smooth(S.fadeT); ctx.drawImage(buf, 0, 0); ctx.globalAlpha = 1; }
    requestAnimationFrame(frame);
  }
  buildStages(); resize(); go(0); S.fadeT = 1; requestAnimationFrame(frame);
